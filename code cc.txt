#include <stdio.h>

int main() {
    int choice, roll;
    char name[50];
    float marks[5], total = 0, avg = 0;
    char grade;

    do {
        printf("\nSTUDENT MARKS MANAGEMENT SYSTEM\n");
        printf("--------------------------------\n");
        printf("1. Enter Student Details\n");
        printf("2. Calculate Result\n");
        printf("3. Display Result\n");
        printf("4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                printf("Enter Roll Number: ");
                scanf("%d", &roll);
                printf("Enter Student Name: ");
                scanf("%s", name);
                total = 0;
                for (int i = 0; i < 5; i++) {
                    printf("Enter Subject %d Mark: ", i + 1);
                    scanf("%f", &marks[i]);
                    total += marks[i];
                }
                break;

            case 2:
                avg = total / 5.0;
                if (avg >= 90) grade = 'A';
                else if (avg >= 75) grade = 'B';
                else if (avg >= 60) grade = 'C';
                else if (avg >= 50) grade = 'D';
                else grade = 'F';
                printf("\nResult Calculated!\n");
                break;

            case 3:
                printf("\nSTUDENT RESULT\n");
                printf("--------------\n");
                printf("Roll Number : %d\n", roll);
                printf("Name        : %s\n", name);
                printf("Total       : %.0f\n", total);
                printf("Average     : %.2f\n", avg);
                printf("Grade       : %c\n", grade);
                printf("Result      : %s\n", avg >= 35 ? "PASS" : "FAIL");
                break;

            case 4:
                printf("Exiting...\n");
                break;

            default:
                printf("Invalid choice!\n");
        }
    } while (choice != 4);

    return 0;
}
